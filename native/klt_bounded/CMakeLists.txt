cmake_minimum_required(VERSION 3.20)
project(klt_bounded C CXX)
# C++ port of solutions/klt_bounded (the e3/e5 configuration). wasm: viobench bundle <this directory> -o x.wasm
set(CMAKE_CXX_STANDARD 17)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()
set(VIOBENCH_NATIVE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../viobench/viobench/solution/runtime/native" CACHE PATH "viobench native runtime directory")
include(${VIOBENCH_NATIVE_DIR}/viobench_reactor.cmake)
viobench_find_opencv(core imgproc video)
if(CMAKE_SYSTEM_NAME STREQUAL "WASI")
    add_compile_options(-msimd128 -fwasm-exceptions "SHELL:-mllvm -wasm-use-legacy-eh=false")
endif()
viobench_solution(solution solution.cpp)
target_include_directories(solution PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${OpenCV_INCLUDE_DIRS})
target_link_libraries(solution PRIVATE ${OpenCV_LIBS})
if(CMAKE_SYSTEM_NAME STREQUAL "WASI")
    target_link_options(solution PRIVATE -fwasm-exceptions -Wl,-z,stack-size=8388608 -Wl,--stack-first)
    target_link_libraries(solution PRIVATE unwind)
endif()
