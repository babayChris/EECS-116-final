cmake_minimum_required(VERSION 3.20)
project(vio_ekf C CXX)
# Compact MSCKF-style VIO without OpenCV. wasm: viobench bundle <this directory> -o x.wasm
set(CMAKE_CXX_STANDARD 17)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()
set(VIOBENCH_NATIVE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../viobench/viobench/solution/runtime/native" CACHE PATH "viobench native runtime directory")
include(${VIOBENCH_NATIVE_DIR}/viobench_reactor.cmake)
if(CMAKE_SYSTEM_NAME STREQUAL "WASI")
    add_compile_options(-msimd128 -O3 ${VIO_EXTRA_FLAGS})
endif()
viobench_solution(solution solution.cpp)
target_include_directories(solution PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(solution PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-fno-exceptions>)
